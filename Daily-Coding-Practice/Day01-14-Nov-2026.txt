# Ansible Task:- Install Nginx on Pathnex Server

---
- name: Install and start Nginx on Pathnex Server
  hosts: all
  become: yes

  tasks:
    - name: Install Nginx
      yum:
        name:   Nginx
        state:  present

# Kubernetes Task -- Create Nginx Pod 

apiVersion: v1
kind: Pod
metadata: 
    name: pathnex-Pod
spec: 
    containers:
        - name: web
          image: nginx:latest
          ports:
              - containerPort: 80

#Git Integration

#Jenkins Pipeline -- Checkout Git

pipeline {
    agent any
        stages {
            stage('Checkout') {
                steps { 
                    git branch: 'main', 
                        url:'https://github.com/Pathnex/sample-java-app.git'
                }
            }
            stage('ListFiles') {
                steps {
                    sh 'ls -la'
                }
            }
        }
}

# GitLab CI -- Checkout Git

stages:
    - checkout
    - listfiles

git-checkout:
    stage: checkout
    script: 
        -git clone https://github.com/Pathnex/sample-java-app.git
        - cd sample-java-app

git-listfiles:
    stage: listfiles
    script: 
        - ls -la


# Create File .github/workflows/git-checkout.yml

name: Git Checkout and List ListFiles

on:
  push:
      branches:
            - main

  workflows_dispatch:

  jobs:
      checkout-and-list-files:
            runs-on: ubuntu-latest

            steps:
                - name: Checkout Repository
                  uses: actions/checkout@v4

                - name: List files
                  run: ls -la
